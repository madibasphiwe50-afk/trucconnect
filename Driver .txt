// ── Image quality pre-check using canvas analysis ──
// Returns { ok: bool, reason: string }
function checkImageQuality(dataUrl) {
  return new Promise(function(resolve) {
    var img = new Image();
    img.onload = function() {
      var w = img.width, h = img.height;
      // Too small — likely cropped or low-res
      if (w < 100 || h < 100) {
        return resolve({ ok: false, reason: 'Image too small or cropped. Please retake the photo closer to the document.' });
      }
      var canvas = document.createElement('canvas');
      var MAX = 300;
      var scale = Math.min(MAX / w, MAX / h, 1);
      canvas.width = Math.round(w * scale);
      canvas.height = Math.round(h * scale);
      var ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
      var pixels = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
      var brightness = 0, contrast = 0, prev = 0;
      var total = pixels.length / 4;
      for (var i = 0; i < pixels.length; i += 4) {
        var lum = (pixels[i] * 0.299 + pixels[i+1] * 0.587 + pixels[i+2] * 0.114);
        brightness += lum;
        contrast += Math.abs(lum - prev);
        prev = lum;
      }
      brightness /= total;
      contrast /= total;
      // Too dark
      if (brightness < 30) {
        return resolve({ ok: false, reason: 'Image is too dark. Please take the photo in better lighting.' });
      }
      // Too bright / washed out
      if (brightness > 230) {
        return resolve({ ok: false, reason: 'Image is overexposed (too bright). Avoid direct flash on the document.' });
      }
      // Very low contrast = blurry or blank
      if (contrast < 3) {
        return resolve({ ok: false, reason: 'Image appears blurry or blank. Please hold the camera steady and retake.' });
      }
      resolve({ ok: true, reason: 'Image quality OK' });
    };
    img.onerror = function() {
      resolve({ ok: false, reason: 'Could not read image file. Please try a different photo.' });
    };
    img.src = dataUrl;
  });
}

// ── Parse a date string extracted by Claude into a JS Date ──
// Handles formats: DD/MM/YYYY, MM/YYYY, YYYY-MM-DD, "January 2026", etc.
function parseExtractedDate(str) {
  if (!str || str === 'NOT_FOUND') return null;
  str = str.trim();
  // DD/MM/YYYY or DD-MM-YYYY
  var m = str.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})$/);
  if (m) return new Date(parseInt(m[3]), parseInt(m[2]) - 1, parseInt(m[1]));
  // MM/YYYY or MM-YYYY
  m = str.match(/^(\d{1,2})[\/\-](\d{4})$/);
  if (m) return new Date(parseInt(m[2]), parseInt(m[1]) - 1, 1);
  // YYYY-MM-DD
  m = str.match(/^(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})$/);
  if (m) return new Date(parseInt(m[1]), parseInt(m[2]) - 1, parseInt(m[3]));
  // "January 2026" or "Jan 2026"
  var months = ['jan','feb','mar','apr','may','jun','jul','aug','sep','oct','nov','dec'];
  m = str.toLowerCase().match(/([a-z]+)\s+(\d{4})/);
  if (m) { var mi = months.indexOf(m[1].slice(0,3)); if (mi !== -1) return new Date(parseInt(m[2]), mi, 1); }
  // "2026" — year only
  m = str.match(/^(\d{4})$/);
  if (m) return new Date(parseInt(m[1]), 11, 31); // end of year
  return null;
}

      var errMsg = '⚠️ Verification failed (server error ' + resp.status + '). Please try again.';
      if (resp.status === 401) errMsg = '⚠️ Verification service misconfigured (invalid API key). Contact support.';
      else if (resp.status === 429) errMsg = '⚠️ Too many attempts. Please wait a moment and try again.';
      return { ok: false, reason: errMsg, status: 'error', expiryDate: null };
    }

    var data = await resp.json();
    var raw = (data.content && data.content[0] && data.content[0].text || '').trim();
    raw = raw.replace(/```json|```/g, '').trim();

    var parsed;
    try { parsed = JSON.parse(raw); } catch(e) {
      return { ok: false, reason: '⚠️ Could not read AI scan result. Please retake the photo and try again.', status: 'error', expiryDate: null };
    }

    // ── Step 4: Wrong document type ──
    if (!parsed.is_correct_doc) {
      return {
        ok: false,
        reason: 'This doesn\'t look like a ' + label + '. Please re-upload the correct document.',
        status: 'wrong_doc',
        expiryDate: null
      };
    }

    var issues = (parsed.image_issues || '').toLowerCase();
    if (issues && issues !== 'ok' && (issues.includes('blurr') || issues.includes('dark') || issues.includes('cropp') || issues.includes('unclear'))) {
      return {
        ok: false,
        reason: '⚠️ Image issue: ' + parsed.image_issues + '. Please retake the photo.',
        status: 'poor_image',
        expiryDate: null
      };
    }

    // ── Step 6: Expiry date extraction & validity check ──
    var expiryStr = parsed.expiry_date || 'NOT_FOUND';
    var expiryDate = parseExtractedDate(expiryStr);
    var today = new Date(); today.setHours(0,0,0,0);

    if (!expiryDate) {
      // Could not read expiry — accept but flag
      var result = {

        status: 'unknown',
        expiryDate: null
      };
      _saveDocVerification(docType, result, expiryStr);
      return result;
    }

    var isExpired = expiryDate < today;
    var expiryFormatted = expiryDate.toLocaleDateString('en-ZA', { day:'2-digit', month:'short', year:'numeric' });

    if (isExpired) {
      var expResult = {
        ok: false,
        reason: '❌ ' + label + ' EXPIRED on ' + expiryFormatted + '. Please renew before continuing.',
        status: 'expired',
        expiryDate: expiryFormatted
      };
      _saveDocVerification(docType, expResult, expiryStr);
      return expResult;
    }

    // Calculate days until expiry for a "expiring soon" warning
    var daysLeft = Math.round((expiryDate - today) / (1000 * 60 * 60 * 24));
    var validMsg = '✅ ' + label + ' VALID — expires ' + expiryFormatted;
    if (daysLeft <= 30) validMsg = '⚠️ ' + label + ' valid but expires soon (' + daysLeft + ' days). Consider renewing.';

    var validResult = {
      ok: true,
      reason: validMsg,
      status: daysLeft <= 30 ? 'expiring_soon' : 'valid',
      expiryDate: expiryFormatted
    };
    _saveDocVerification(docType, validResult, expiryStr);
    return validResult;

  } catch(e) {
    return { ok: false, reason: '⚠️ Network error during verification. Check your connection and try again.', status: 'error', expiryDate: null };
  }
}

      FB.updateDoc(FB.doc(FB.db, 'drivers', uid), { fcmToken: token }).catch(function(){});
    }

    // Handle notifications while the app is OPEN (foreground)
    FB.onMessage(FB.messaging, function(payload) {
      var n = payload.notification || {};
      var title = n.title || 'TruckConnect';
      var body  = n.body  || '';
      // Show as an in-app toast instead of a browser notification
      var t = document.getElementById('toast');

    console.warn('TruckConnect FCM init failed:', e.message || e);
  }
}

function validateSAID(id){
  id=(id||'').replace(/\D/g,'');
  if(id.length!==13)return false;
  var yy=parseInt(id.substring(0,2),10);
  var mm=parseInt(id.substring(2,4),10);
  var dd=parseInt(id.substring(4,6),10);
  var yyyy=(yy<=25?2000:1900)+yy;
  var dob=new Date(yyyy,mm-1,dd);
  if(dob.getFullYear()!==yyyy||dob.getMonth()!==mm-1||dob.getDate()!==dd)return false;
  var now=new Date();
  if(dob>=now||yyyy<now.getFullYear()-120)return false;
  var sum=0;
  for(var i=0;i<13;i++){
    var digit=parseInt(id[i],10);
    if(i%2===0){sum+=digit;}
    else{var d2=digit*2;sum+=d2>9?d2-9:d2;}
  }
  return sum%10===0;
}

function handleGoogleAuth(mode){
  var FB=window._FB;if(!FB){toast('Firebase not ready. Please refresh.');return;}
  if(typeof google==='undefined'||!google.accounts){toast('Google loading, please try again.');return;}
  // Use login_hint if we have a remembered Google account so it auto-selects
  var loginHint='';
  try{
    var rem=JSON.parse(localStorage.getItem('tc_remembered_account')||'null');
    if(rem&&rem.isGoogle&&rem.email) loginHint=rem.email;
  }catch(e){}
  var tokenClient=google.accounts.oauth2.initTokenClient({
    client_id:'798935050164-4i8k63gsti7gqnjofq3329050ap4tq34.apps.googleusercontent.com',
    scope:'email profile openid',
    hint:loginHint||undefined,
    callback:function(response){
      if(response.error){toast('Google sign-in failed. Please try again.');return;}
      var credential=FB.GoogleAuthProvider.credential(null,response.access_token);
      FB.signInWithCredential(FB.auth,credential)
      .then(function(result){
        var gUser=result.user;
        if(!gUser.emailVerified){
          toast('⚠ This Google account is not verified. Please use a real account.');
          FB.signOut(FB.auth); return;
        }
        currentUid=gUser.uid;
        // Persist Google account hint so next sign-in auto-selects this account
        try{
          var existing=JSON.parse(localStorage.getItem('tc_remembered_account')||'{}');
          existing.isGoogle=true;
          existing.email=gUser.email||existing.email||'';
          existing.name=gUser.displayName||existing.name||'';
          existing.photo=gUser.photoURL||existing.photo||'';
          localStorage.setItem('tc_remembered_account',JSON.stringify(existing));
        }catch(e){}
        return FB.getDoc(FB.doc(FB.db,'drivers',currentUid)).then(function(snap){
          if(snap.exists()){
            dData=snap.data();
            dData._wasGoogle=true;
            saveSession();
            if(dData.status==='approved')goToDriverHome();
            else{goTo('s-pending');txt('pend-sub-val','✅ TruckConnect Plan (R300/mo)');}
          } else {
            // Fallback: search by email
            return FB.getDocs(FB.query(FB.collection(FB.db,'drivers'),FB.where('email','==',gUser.email||''))).then(function(qsnap){
              if(!qsnap.empty){
                dData=qsnap.docs[0].data();
                dData._wasGoogle=true;
                saveSession();
                if(dData.status==='approved')goToDriverHome();
                else{goTo('s-pending');txt('pend-sub-val','✅ TruckConnect Plan (R300/mo)');}
              } else {
                dData.email=gUser.email||'';
                dData.name=gUser.displayName||'';
                dData._googleAuth=true;
                dData._wasGoogle=true;
                dStep=1;
                goTo('s-auth');switchTab('signup');
                renderStep();
                toast('Google account connected! Complete your ID details.');
              }
            });
          }
        });
      })
      .catch(function(err){
        var msg='Google sign-in failed. Please try again.';
        if(err.code==='auth/network-request-failed')msg='No internet connection.';
        showErr('si-err',msg);toast(msg);
      });
    }
  });
  // 'none' prompt = silent if hint matches; 'select_account' if no hint
  tokenClient.requestAccessToken({prompt: loginHint ? 'none' : 'select_account'});
}

function createAccount(){

function initDriverMap(){
  if(driverMap)return;
  function _init(){
    if(typeof google==='undefined'||!google.maps||!google.maps.Map){setTimeout(_init,300);return;}

    driverMap=new google.maps.Map(document.getElementById('driver-map'),{
      center:{lat:-26.2041,lng:28.0473},
      zoom:14,
      mapTypeId:'roadmap',
      disableDefaultUI:true,
      styles:_isDarkMode?MAP_STYLES_DARK:MAP_STYLES_LIGHT
    });
    // Force map to fill its container
    setTimeout(function(){google.maps.event.trigger(driverMap,'resize');},200);

    // ── Place / update driver dot ──
    function _placeDriverMarker(latlng){
      if(driverMarker){
        driverMarker.setPosition(latlng);
      } else {
        driverMarker=new google.maps.Marker({
          position:latlng,
          map:driverMap,
          icon:{
            path:google.maps.SymbolPath.CIRCLE,
            scale:9,
            fillColor:'#1c1c1e',
            fillOpacity:1,
            strokeColor:'#ffffff',
            strokeWeight:3
          },
          title:'You are here',
          zIndex:999
        });
      }
    }

    // ── Step 1: get position immediately so map centres on driver ──
    function _startGPS(){
      if(!navigator.geolocation){
        toast('GPS not available on this device.');
        return;
      }
      // One-shot first fix — centres the map right away
      navigator.geolocation.getCurrentPosition(
        function(pos){

  if(!jobTruck)return true;

  if(j===d)return true;
  var order=['1ton','bakkie','1 ton','small','3ton','3 ton','medium','5ton','5 ton','8ton','8 ton','large','14ton','14 ton','rigid','34ton','34 ton','semi','extra large'];
  var ji=order.findIndex(function(x){return j.indexOf(x)!==-1;});

  if(ji===-1||di===-1)return true;
  return di>=ji;
}

function _parsePriceNum(j){
  if(j.priceNum)return j.priceNum;

function _applyFilter(jobs,filter){
  if(filter==='all')return jobs;
  if(filter==='today')return jobs.filter(function(j){return j.urgent||(j.date||'').toLowerCase()==='today'||(j.date||'').toLowerCase()==='asap';});
  if(filter==='highpay')return jobs.filter(function(j){return _parsePriceNum(j)>=3000;});
  if(filter==='short')return jobs.filter(function(j){return(j.distKm||0)<400;});
  if(filter==='long')return jobs.filter(function(j){return(j.distKm||0)>=400;});
  return jobs;
}

function _truckMatchLevel(jobTruck,driverTruck){
  // Returns 'perfect', 'compat', or 'none'
  if(!jobTruck)return'compat';
  var j=(jobTruck||'').toLowerCase().replace(/\s+/g,'');
  var d=(driverTruck||'').toLowerCase().replace(/\s+/g,'');
  // Exact match
  if(j===d||d.indexOf(j)!==-1||j.indexOf(d)!==-1)return'perfect';
  // Capacity order — driver can always take jobs equal to or smaller than their truck
  var order=[
    ['1ton','bakkie'],
    ['3ton'],
    ['5ton'],
    ['8ton'],
    ['14ton','rigid'],
    ['34ton','semi','extralarge']
  ];
  var jRank=-1,dRank=-1;
  order.forEach(function(grp,i){
    grp.forEach(function(kw){if(j.indexOf(kw)!==-1)jRank=i;if(d.indexOf(kw)!==-1)dRank=i;});
  });
  if(jRank===-1||dRank===-1)return'compat'; // unknown — allow
  if(dRank===jRank)return'perfect';
  if(dRank>jRank)return'compat'; // driver's truck is bigger — can carry
  return'none'; // job requires bigger truck than driver has
}

        var timeStr=mins<60?mins+' min':Math.floor(mins/60)+'h '+(mins%60)+'m';
        if(legLabel==='leg1'){
          txt('jp-leg1-info','📍 '+km+' km · '+timeStr+' to pickup');
          // Calculate ETA clock time at pickup
          var nowMs=Date.now();
          var etaMs=nowMs+(r.duration*1000);
          var etaDate=new Date(etaMs);
          var eh=etaDate.getHours(),em=etaDate.getMinutes();
          var eap=eh>=12?'PM':'AM';eh=eh%12||12;
          txt('jp-eta-pickup',eh+':'+(em<10?'0'+em:em)+' '+eap);
        }
        // Draw animated polyline on Google Maps
        if(typeof google!=='undefined'&&driverMap){
          var path=r.geometry.coordinates.map(function(c){return{lat:c[1],lng:c[0]};});
          if(animated){
            // Animated dashed line for leg1 (driver → pickup)
            var lineSymbol={path:'M 0,-1 0,1',strokeOpacity:1,scale:3};
            var poly=new google.maps.Polyline({
              path:path,
              strokeOpacity:0,
              icons:[{icon:lineSymbol,offset:'0',repeat:'16px'}],
              strokeColor:color,
              map:driverMap
            });
            _jpPolylines.push(poly);
            // Animate the dash
            var count=0;
            var animId=setInterval(function(){
              count=(count+1)%200;
              var icons=poly.get('icons');
              icons[0].offset=(count/2)+'%';
              poly.set('icons',icons);
            },40);
            poly._animId=animId;
          } else {
            // Solid bold line for leg2 (pickup → dropoff)
            var poly2=new google.maps.Polyline({
              path:path,
              strokeColor:color,
              strokeOpacity:0.95,
              strokeWeight:6,
              map:driverMap
            });
            _jpPolylines.push(poly2);
          }
        }
      }
    }).catch(function(){
      // fallback straight line
      if(typeof google!=='undefined'&&driverMap){
        var straightPath=[{lat:from[1],lng:from[0]},{lat:to[1],lng:to[0]}];
        var fb=new google.maps.Polyline({path:straightPath,strokeColor:color,strokeOpacity:0.6,strokeWeight:4,map:driverMap});
        _jpPolylines.push(fb);
      }
    });
  }

  if(driverPos&&pickupCoords&&pickupCoords[0]){
    // Leg 1: driver → pickup (animated dashed grey)
    fetchAndDraw(driverPos,pickupCoords,'#9ca3af','leg1',true);
  } else {
    txt('jp-leg1-info','');
  }
  if(pickupCoords&&pickupCoords[0]&&dropoffCoords&&dropoffCoords[0]){
    // Leg 2: pickup → dropoff (solid bold dark)
    fetchAndDraw(pickupCoords,dropoffCoords,'#1c1c1e','leg2',false);
  }

  // Fit map to show all points, with delay for popup animation
  setTimeout(function(){_fitJobBounds(fitPoints);},350);
}

var _jpClockInterval=null;
function showJobPopup(b){
  pendingJob=b;
  // Populate stats
  txt('jp-meta',(b.cargo||'General Freight')+' · '+(b.date||'ASAP'));
  txt('jp-dist',b.dist||'—');
  txt('jp-earn',b.price||'—');
  txt('jp-cargo',b.cargo||'—');
  txt('jp-from-addr',b.from||'—');
  txt('jp-to-addr',b.to||'—');
  txt('jp-leg1-info','Calculating...');
  txt('jp-eta-pickup','Calculating...');
  // Live clock
  if(_jpClockInterval)clearInterval(_jpClockInterval);
  function _updateJpClock(){
    var now=new Date();
    var h=now.getHours(),m=now.getMinutes();
    var ap=h>=12?'PM':'AM';h=h%12||12;
    txt('jp-current-time',h+':'+(m<10?'0'+m:m)+' '+ap);
  }
  _updateJpClock();
  _jpClockInterval=setInterval(_updateJpClock,15000);
  $('job-popup').style.transform='translateY(0)';

  // ── Resolve coords first, THEN draw the 3-stop route ──
  // This ensures geocoding happens even before the job is accepted
  function _doDrawRoute(){
    // Get fresh driver GPS position
    if(navigator.geolocation){
      navigator.geolocation.getCurrentPosition(
        function(pos){driverLat=pos.coords.latitude;driverLng=pos.coords.longitude;_drawJobRequestRoute(b);},
        function(){_drawJobRequestRoute(b);},
        {enableHighAccuracy:true,timeout:6000,maximumAge:10000}
      );
    } else {
      _drawJobRequestRoute(b);
    }
  }

  // If coords already present, draw immediately
  if(b.fromCoords&&b.fromCoords[0]&&b.toCoords&&b.toCoords[0]){
    setTimeout(_doDrawRoute, 150);
  } else {
    // Geocode missing addresses, then draw
    var tasks=[];
    if(!b.fromCoords&&b.from) tasks.push(_geocodeAddress(b.from).then(function(c){if(c)b.fromCoords=c;}));
    if(!b.toCoords&&b.to)     tasks.push(_geocodeAddress(b.to).then(function(c){if(c)b.toCoords=c;}));
    Promise.all(tasks).then(function(){
      // Also fetch driving distance now that we have coords
      if(b.fromCoords&&b.toCoords&&!b.dist){
        var osrmUrl='https://router.project-osrm.org/route/v1/driving/'+
          b.fromCoords[0]+','+b.fromCoords[1]+';'+
          b.toCoords[0]+','+b.toCoords[1]+'?overview=false';
        fetch(osrmUrl).then(function(r){return r.json();}).then(function(d){
          if(d.routes&&d.routes[0]){
            var km=(d.routes[0].distance/1000).toFixed(1);
            b.dist=km+' km';
            txt('jp-dist',b.dist);
          }
        }).catch(function(){});
      }
      setTimeout(_doDrawRoute, 150);
    });
  }
}

function acceptJob(){
  if(!pendingJob)return;
  var b=pendingJob;
  var FB=window._FB;
  if(FB){
    // Use transaction to prevent double-booking
    FB.runTransaction(FB.db, function(transaction) {
      var docRef = FB.doc(FB.db, 'bookings', b.id);
      return transaction.get(docRef).then(function(snap) {
        if (!snap.exists() || snap.data().ds !== 'avail') {
          throw new Error('Job already taken');
        }
        transaction.update(docRef, {
          ds: 'accepted', driverUid: currentUid, driverName: dData.name,
          driverTruck: dData.truckType || dData.truck || '',
          driverPhone: dData.phone || '',
          acceptedAt: new Date().toISOString()
        });
      });
    })
    .then(function(){
      b.ds='accepted';b.driverUid=currentUid;b.driverName=dData.name;
      dTrips++;dTodayEarn+=(b.priceNum||0);dWeekEarn+=(b.priceNum||0);
      txt('earn-today','R'+dTodayEarn.toLocaleString());txt('earn-week','R'+dWeekEarn.toLocaleString());txt('earn-trips',dTrips);
      acceptedJobs.push(Object.assign({},b,{at:new Date().toLocaleTimeString()}));
      renderJobs();hideJobPopup();toast('Job accepted! 🎉 Open My Jobs to start.');
      // ── Resolve coords then broadcast ETA ──
      var _accepted=acceptedJobs[acceptedJobs.length-1];
      _resolveJobCoords(_accepted).then(function(){
        return _getDriverPosition();
      }).then(function(driverPos){
        if(driverPos&&_accepted.toCoords) _broadcastETA(b.id, driverPos, _accepted.toCoords);
      });
    })
    .catch(function(e){
      if(e.message==='Job already taken') {
        toast('❌ Job already taken.'); hideJobPopup();
      } else {
        toast('Could not accept: '+e.message);
      }
    });
  } else {
    dTrips++;dTodayEarn+=(b.priceNum||0);dWeekEarn+=(b.priceNum||0);
    txt('earn-today','R'+dTodayEarn.toLocaleString());txt('earn-week','R'+dWeekEarn.toLocaleString());txt('earn-trips',dTrips);
    acceptedJobs.push(Object.assign({},b,{at:new Date().toLocaleTimeString()}));
    renderJobs();hideJobPopup();toast('Job accepted! 🎉');
  }
}

window.startTrip=function(jobId){
  var FB=window._FB;
  if(FB){
    FB.updateDoc(FB.doc(FB.db,'bookings',jobId),{ds:'in_progress',startedAt:new Date().toISOString()})
      .then(function(){
        var j=acceptedJobs.find(function(x){return x.id===jobId;});
        if(j)j.ds='in_progress';
        renderJobs();toast('🚛 Trip started! Safe travels.');
      })
      .catch(function(e){toast('Error: '+e.message);});
  } else {
    var j=acceptedJobs.find(function(x){return x.id===jobId;});
    if(j)j.ds='in_progress';
    renderJobs();toast('🚛 Trip started!');
  }
};

window.completeTrip=function(jobId){
  var FB=window._FB;
  if(FB){
    FB.updateDoc(FB.doc(FB.db,'bookings',jobId),{ds:'completed',completedAt:new Date().toISOString()})
      .then(function(){
        acceptedJobs=acceptedJobs.filter(function(x){return x.id!==jobId;});
        renderJobs();toast('✅ Delivery complete! Great work.');
      })
      .catch(function(e){toast('Error: '+e.message);});
  } else {
    acceptedJobs=acceptedJobs.filter(function(x){return x.id!==jobId;});
    renderJobs();toast('✅ Delivery complete!');
  }
};

window.cancelTrip=function(jobId){
  if(!confirm('Return this job to pending? Another driver will be able to accept it.'))return;
  var FB=window._FB;
  if(FB){
    FB.updateDoc(FB.doc(FB.db,'bookings',jobId),{ds:'avail',driverUid:null,driverName:null,driverTruck:null,acceptedAt:null})
      .then(function(){
        acceptedJobs=acceptedJobs.filter(function(x){return x.id!==jobId;});
        renderJobs();toast('Job returned to available pool.');
      })
      .catch(function(e){toast('Error: '+e.message);});
  } else {
    acceptedJobs=acceptedJobs.filter(function(x){return x.id!==jobId;});
    renderJobs();toast('Job cancelled.');
  }
};

// ── Clock ticker ──
var _navClockInterval=null;
function startNavClock(){
  function tick(){
    var now=new Date();
    var h=now.getHours(),m=now.getMinutes();
    var ampm=h>=12?'PM':'AM';
    h=h%12||12;
    var el=document.getElementById('nav-clock');
    if(el)el.textContent=h+':'+(m<10?'0'+m:m)+' '+ampm;
  }
  tick();
  _navClockInterval=setInterval(tick,10000);
}
function stopNavClock(){
  if(_navClockInterval){clearInterval(_navClockInterval);_navClockInterval=null;}
}

function startNav(from,to){
  var url='https://router.project-osrm.org/route/v1/driving/'+from[0]+','+from[1]+';'+to[0]+','+to[1]+'?overview=full&geometries=geojson&steps=true';
  fetch(url).then(function(r){return r.json();}).then(function(d){
    if(!d.routes||!d.routes.length){toast('Could not load navigation route.');return;}
    var route=d.routes[0];
    var steps=(route.legs[0]||{}).steps||[];
    var currentStepIdx=0;
    var routeCoords=route.geometry.coordinates;

    $('ov-navigation').style.display='flex';
    startNavClock();

    // Distance between two [lng,lat] points in metres
    function mDist(a,b){
      var R=6371000,toR=Math.PI/180;
      var dLat=(b[1]-a[1])*toR,dLng=(b[0]-a[0])*toR;
      var aa=Math.sin(dLat/2)*Math.sin(dLat/2)+Math.cos(a[1]*toR)*Math.cos(b[1]*toR)*Math.sin(dLng/2)*Math.sin(dLng/2);
      return R*2*Math.atan2(Math.sqrt(aa),Math.sqrt(1-aa));
    }
    // Find nearest route coord index to driver position
    function nearestIdx(pos){
      var best=0,bestD=Infinity;
      for(var i=0;i<routeCoords.length;i++){
        var dd=mDist(pos,routeCoords[i]);
        if(dd<bestD){bestD=dd;best=i;}
      }
      return best;
    }
    // Remaining distance from a coord index to end
    function remainingDist(fromIdx){
      var total=0;
      for(var i=fromIdx;i<routeCoords.length-1;i++) total+=mDist(routeCoords[i],routeCoords[i+1]);
      return total;
    }
    // Show turn instruction for step
    function showStep(idx){
      if(!steps[idx])return;
      var s=steps[idx];
      var instr=s.maneuver&&s.maneuver.instruction?s.maneuver.instruction:'Continue';
      txt('nav-instruction',instr);
      var d2=s.distance;
      var distText=d2>=1000?(d2/1000).toFixed(1)+' km':Math.round(d2)+' m';
      txt('nav-distance-next',distText);
      // Speak the instruction with distance
      _speak('In '+distText+', '+instr);
      // Turn arrow
      var manType=s.maneuver?s.maneuver.type:'';
      var manMod=s.maneuver?s.maneuver.modifier:'';
      var arrows={'turn left':'↰','turn right':'↱','turn sharp left':'↰','turn sharp right':'↱','turn slight left':'↖','turn slight right':'↗','continue':'⬆','merge':'⬆','roundabout':'↻','rotary':'↻','u-turn':'↩','arrive':'📍','depart':'⬆'};
      var key=(manType+' '+manMod).trim().toLowerCase();
      var arrow=arrows[key]||arrows[manType]||'⬆';
      var el=document.getElementById('nav-turn-icon');
      if(el)el.textContent=arrow;
    }
    // Draw blue route line
    function drawRoute(){
      if(!navMap)return;
      if(navMap.getSource('route')){
        navMap.getSource('route').setData({type:'Feature',geometry:route.geometry});
      } else {
        navMap.addSource('route',{type:'geojson',data:{type:'Feature',geometry:route.geometry}});
        navMap.addLayer({id:'route',type:'line',source:'route',layout:{'line-join':'round','line-cap':'round'},paint:{'line-color':'#4b3fbd','line-width':8,'line-opacity':1}});
      }
      new mapboxgl.Marker({color:'#4ade80'}).setLngLat(from).addTo(navMap);
      new mapboxgl.Marker({color:'#ef4444'}).setLngLat(to).addTo(navMap);
    }

    if(!navMap){
      var navStyle=_isDarkMode?'https://basemaps.cartocdn.com/gl/dark-matter-gl-style/style.json':'https://tiles.openfreemap.org/styles/liberty';
      navMap=new mapboxgl.Map({container:'nav-map-tiles',style:navStyle,center:from,zoom:15,pitch:50,attributionControl:false});
      navMap.on('load',function(){drawRoute();});
    } else {
      drawRoute();
    }

    showStep(0);
    var startDist=route.distance>=1000?(route.distance/1000).toFixed(1)+' km':Math.round(route.distance)+' m';
    _speak('Navigation started. Your destination is '+startDist+' away.');
    var initDist=route.distance/1000,initDur=route.duration;
    var initMins=Math.round(initDur/60);
    var initHrs=Math.floor(initMins/60);
    txt('nav-eta',(initHrs>0?initHrs+'h '+(initMins%60)+'m':initMins+' min')+' • '+initDist.toFixed(1)+' km');

    if(navigator.geolocation)navWatchId=navigator.geolocation.watchPosition(function(pos){
      var speed=Math.round((pos.coords.speed||0)*3.6);
      txt('nav-speed',speed);
      var driverPos=[pos.coords.longitude,pos.coords.latitude];
      if(navMap)navMap.flyTo({center:driverPos,bearing:pos.coords.heading||0,pitch:50,zoom:16});

      // Live remaining distance
      var ni=nearestIdx(driverPos);
      var remKm=remainingDist(ni)/1000;

      // Live ETA
      var avgKmh=speed>5?speed:40;
      var etaMins=Math.round((remKm/avgKmh)*60);
      var etaHrs=Math.floor(etaMins/60);
      txt('nav-eta',(etaHrs>0?etaHrs+'h '+(etaMins%60)+'m':etaMins+' min')+' • '+remKm.toFixed(1)+' km');

      // Advance step
      if(steps[currentStepIdx+1]){
        var nextManLoc=steps[currentStepIdx+1].maneuver.location;
        if(mDist(driverPos,nextManLoc)<30){currentStepIdx++;showStep(currentStepIdx);}
      }
    },{enableHighAccuracy:true,maximumAge:2000,timeout:10000});
  }).catch(function(e){toast('Navigation error: '+e.message);});
}
function stopNav(){
  if(navWatchId){navigator.geolocation.clearWatch(navWatchId);navWatchId=null;}
  stopNavClock();
  // Close music panel if open
  var panel=document.getElementById('nav-music-panel');
  if(panel)panel.classList.remove('open');
  if(navMap){navMap.remove();navMap=null;}
  $('ov-navigation').style.display='none';
}

// ─── Earnings history store (in-memory, seeded with sample data) ───
var earningsHistory = [];
var _earnPeriod = 'day';

function _seedEarningsHistory() {
  // Seed last 14 days of sample data so the chart has something to show
  var today = new Date();
  earningsHistory = [];
  for (var i = 13; i >= 0; i--) {
    var d = new Date(today);
    d.setDate(d.getDate() - i);
    var trips = Math.floor(Math.random() * 4);
    var earn = trips * (Math.floor(Math.random() * 2500) + 500);
    earningsHistory.push({
      date: d.toLocaleDateString('en-ZA', {day:'2-digit', month:'short'}),
      dateObj: d,
      trips: trips,
      earned: earn
    });
  }
  // Today's real earnings override the last entry
  earningsHistory[earningsHistory.length - 1].earned = dTodayEarn;
  earningsHistory[earningsHistory.length - 1].trips = dTrips;
}

window._setEarnPeriod = function(period, btn) {
  _earnPeriod = period;
  document.querySelectorAll('.earn-period-tab').forEach(function(b){b.classList.remove('active');});
  if (btn) btn.classList.add('active');
  _renderEarningsSheet();
};

function _renderEarningsSheet() {
  var today = new Date();
  var hist = earningsHistory;
  var subset, label, chartTitle, bars;

  if (_earnPeriod === 'day') {
    // Show today's trips as hourly breakdown (mock)
    label = 'Total Earned Today';
    chartTitle = 'Today\'s Earnings (by trip)';
    var total = dTodayEarn;
    txt('total-earned', 'R' + total.toLocaleString());
    txt('earn-period-label', label);
    txt('earn-trips-count', dTrips);
    txt('earn-avg-trip', dTrips ? 'R' + Math.round(dTodayEarn / dTrips).toLocaleString() : 'R0');
    txt('earn-chart-title', chartTitle);
    // Bar chart: last 7 trips or hours
    var hours = ['6am','8am','10am','12pm','2pm','4pm','6pm'];
    var vals = hours.map(function(_, i) { return i < dTrips ? Math.floor(dTodayEarn / Math.max(dTrips,1)) : 0; });
    _renderEarnBars(hours, vals, 6);
    _renderEarnHistList([{date:'Today', trips: dTrips, earned: dTodayEarn}]);
  } else if (_earnPeriod === 'week') {
    label = 'Total Earned This Week';
    chartTitle = 'Daily Earnings — This Week';
    var weekHist = hist.slice(-7);
    var weekTotal = weekHist.reduce(function(s,d){return s+d.earned;},0);
    var weekTrips = weekHist.reduce(function(s,d){return s+d.trips;},0);
    txt('total-earned', 'R' + weekTotal.toLocaleString());
    txt('earn-period-label', label);
    txt('earn-trips-count', weekTrips);
    txt('earn-avg-trip', weekTrips ? 'R' + Math.round(weekTotal / weekTrips).toLocaleString() : 'R0');
    txt('earn-chart-title', chartTitle);
    _renderEarnBars(weekHist.map(function(d){return d.date;}), weekHist.map(function(d){return d.earned;}), 6);
    _renderEarnHistList(weekHist.slice().reverse());
  } else {
    label = 'Total Earned This Month';
    chartTitle = 'Daily Earnings — This Month';
    var monthHist = hist.slice(-14);
    var monthTotal = monthHist.reduce(function(s,d){return s+d.earned;},0);
    var monthTrips = monthHist.reduce(function(s,d){return s+d.trips;},0);
    txt('total-earned', 'R' + monthTotal.toLocaleString());
    txt('earn-period-label', label);
    txt('earn-trips-count', monthTrips);
    txt('earn-avg-trip', monthTrips ? 'R' + Math.round(monthTotal / monthTrips).toLocaleString() : 'R0');
    txt('earn-chart-title', chartTitle);
    _renderEarnBars(monthHist.map(function(d){return d.date;}), monthHist.map(function(d){return d.earned;}), 13);
    _renderEarnHistList(monthHist.slice().reverse());
  }
  var nd = new Date(); nd.setDate(nd.getDate() + 14);
  txt('next-payout-date', nd.toLocaleDateString('en-ZA'));
  txt('sub-cost', 'R' + (dData.subscription === 'standard' ? 899 : 699).toLocaleString());
  _checkFCMBanner();
}

function _renderEarnBars(labels, values, todayIdx) {
  var area = $('earn-bar-area'); if (!area) return;
  var max = Math.max.apply(null, values) || 1;
  area.innerHTML = labels.map(function(lbl, i) {
    var pct = Math.round((values[i] / max) * 100);
    var isToday = (i === todayIdx);
    return '<div class="earn-bar-col">' +
      '<div class="earn-bar-fill' + (isToday ? ' today' : '') + '" style="height:' + Math.max(pct, 3) + '%;"></div>' +
      '<div class="earn-bar-x">' + lbl + '</div>' +
    '</div>';
  }).join('');
}

function _renderEarnHistList(rows) {
  var list = $('earn-hist-list'); if (!list) return;
  if (!rows.length) { list.innerHTML = '<div style="padding:20px;text-align:center;color:var(--muted);font-size:13px;">No earnings recorded yet.</div>'; return; }
  list.innerHTML = rows.map(function(r) {
    return '<div class="earn-hist-row">' +
      '<div><div class="earn-hist-date">' + r.date + '</div><div class="earn-hist-trips">' + r.trips + ' trip' + (r.trips !== 1 ? 's' : '') + '</div></div>' +
      '<div class="earn-hist-amount">' + (r.earned ? 'R' + r.earned.toLocaleString() : '—') + '</div>' +
    '</div>';
  }).join('');
}

function _broadcastETA(jobId, fromCoords, toCoords) {
  if (!fromCoords || !toCoords || !toCoords[0] || !toCoords[1]) return;

  fetch(url)
    .then(function(r) { return r.json(); })
    .then(function(d) {
      if (!d.routes || !d.routes.length) return;
      var etaMins = Math.round(d.routes[0].duration / 60);
      var etaTime = new Date(Date.now() + d.routes[0].duration * 1000).toISOString();
      var FB = window._FB;
      if (FB && jobId) {
        FB.updateDoc(FB.doc(FB.db, 'bookings', jobId), {
          driverEtaMins: etaMins,
          driverEtaTime: etaTime,
          driverLat: driverLat || null,
          driverLng: driverLng || null,
          etaUpdatedAt: new Date().toISOString()
        }).catch(function(){});
      }
    })
    .catch(function(){});
}

// COORD RESOLVER — geocodes missing from/toCoords using Google Maps,
// then recalculates real driving distance so display always matches navigation
// ═══════════════════════════════════════════════════════════════

// Nominatim geocoding — free, no API key
function _geocodeAddress(address){
  var query=address.toLowerCase().includes('south africa')?address:address+', South Africa';
  return fetch('https://nominatim.openstreetmap.org/search?format=json&q='+encodeURIComponent(query)+'&countrycodes=za&limit=1',{headers:{'Accept-Language':'en'}})
    .then(function(r){return r.json();})
    .then(function(d){
      if(d&&d.length){return[parseFloat(d[0].lon),parseFloat(d[0].lat)];}
      return null;
    })
    .catch(function(){return null;});
}

function _getDriverPosition(){
  return new Promise(function(resolve){
    if(navigator.geolocation){
      navigator.geolocation.getCurrentPosition(
        function(pos){
          driverLat=pos.coords.latitude;driverLng=pos.coords.longitude;
          resolve([driverLng,driverLat]);
        },
        function(){
          // Fall back to last known, never use hardcoded city
          resolve(driverLat&&driverLng?[driverLng,driverLat]:null);
        },
        {enableHighAccuracy:true,timeout:8000,maximumAge:15000}
      );
    } else {
      resolve(driverLat&&driverLng?[driverLng,driverLat]:null);
    }
  });
}

function _resolveJobCoords(job){
  var tasks=[];
  if(!job.fromCoords&&job.from) tasks.push(_geocodeAddress(job.from).then(function(c){if(c)job.fromCoords=c;}));
  if(!job.toCoords&&job.to)     tasks.push(_geocodeAddress(job.to).then(function(c){if(c)job.toCoords=c;}));
  return Promise.all(tasks).then(function(){
    if(job.fromCoords&&job.toCoords){
      var url='https://router.project-osrm.org/route/v1/driving/'+
        job.fromCoords[0]+','+job.fromCoords[1]+';'+
        job.toCoords[0]+','+job.toCoords[1]+'?overview=false';
      return fetch(url)
        .then(function(r){return r.json();})
        .then(function(d){
          if(d.routes&&d.routes.length){
            var km=(d.routes[0].distance/1000).toFixed(0);
            job.dist=km+' km';
            renderJobs();
          }
        })
        .catch(function(){});
    }
  });
}

  var driverMap = null;
  function initMap(){
    if(driverMap) return;
    driverMap = L.map('driver-map', { zoomControl: false }).setView([-26.2041, 28.0473], 12);
    L.tileLayer('https://tiles.stadiamaps.com/tiles/alidade_smooth/{z}/{x}/{y}{r}.png', {

      attribution: '&copy; Stadia Maps'
    }).addTo(driverMap);

  function _showRoute(from, to){
    initMap();
    // Clear previous layers
    driverMap.eachLayer(function(layer){ if(layer instanceof L.Polyline || layer instanceof L.Marker) driverMap.removeLayer(layer); });
    
    var url = 'https://router.project-osrm.org/route/v1/driving/' + from[0] + ',' + from[1] + ';' + to[0] + ',' + to[1] + '?overview=full&geometries=geojson';
    fetch(url).then(function(r){return r.json();}).then(function(data){
      if(data.code === 'Ok'){
        var route = data.routes[0];
        var coords = route.geometry.coordinates.map(function(c){return [c[1], c[0]];});
        var poly = L.polyline(coords, {color: '#16a34a', weight: 5, opacity: 0.8}).addTo(driverMap);
        L.marker([from[1], from[0]]).addTo(driverMap);
        L.marker([to[1], to[0]]).addTo(driverMap);
        driverMap.fitBounds(poly.getBounds(), {padding: [50, 50]});
      }
    });
  }
  window._showRoute = _showRoute;

  function _loadRememberedAccount(){
    try{
      var rem=JSON.parse(localStorage.getItem('tc_remembered_account')||'null');
      if(!rem||!rem.email) return;
      var card=$('remembered-account-card');
      var fields=$('si-fields-wrap');

      var avEl=$('rem-av');
      if(rem.photo){
        avEl.innerHTML='<img src="'+rem.photo+'" style="width:100%;height:100%;object-fit:cover;border-radius:50%;"/>';
      } else {
        avEl.textContent=(rem.name?rem.name[0].toUpperCase():'?');
      }
      var nameEl=$('rem-name'); if(nameEl) nameEl.textContent=rem.name||rem.email;
      var emailEl=$('rem-email'); if(emailEl) emailEl.textContent=rem.email;
      // Show the right login action inside the card
      var gWrap=$('rem-google-btn-wrap');
      var pWrap=$('rem-pass-wrap');
      if(rem.isGoogle){
        if(gWrap) gWrap.style.display='block';
        if(pWrap) pWrap.style.display='none';
      } else {
        if(gWrap) gWrap.style.display='none';
        if(pWrap) pWrap.style.display='block';
        // Pre-fill email in the hidden standard fields too
        var siEmail=$('si-email'); if(siEmail) siEmail.value=rem.email;
      }
      // Show card, hide standard fields
      card.style.display='block';
      fields.style.display='none';
    }catch(e){}

  on('signin-btn','click',function(){
    // Security: rate-limit failed sign-in attempts
    var now=Date.now();
    try{
      var attempts=JSON.parse(localStorage.getItem('tc_d_signin_attempts')||'[]');
      attempts=attempts.filter(function(t){return now-t<60000;}); // last 60s
      if(attempts.length>=5){
        var wait=Math.ceil((60000-(now-attempts[0]))/1000);
        showErr('si-err','Too many failed attempts. Please wait '+wait+' seconds.');return;
      }
    }catch(e){}
    var e=val('si-email').trim(),p=val('si-pass');
    hideErr('si-email-err');hideErr('si-pass-err');hideErr('si-err');
    if(!e||!e.includes('@')){showErr('si-email-err','Please enter a valid email address');return;}

    var btn=$('signin-btn');btn.disabled=true;btn.textContent='Signing In...';
    var FB=window._FB;
    FB.signInWithEmailAndPassword(FB.auth,e,p)
    .then(function(uc){
      // Clear fail counter on success
      try{localStorage.removeItem('tc_d_signin_attempts');}catch(ex){}
      currentUid=uc.user.uid;
      return FB.getDoc(FB.doc(FB.db,'drivers',currentUid));
    })
    .then(function(snap){
      btn.disabled=false;btn.textContent='Sign In →';

    })
    .catch(function(err){

    });

  // navigateTo — always uses driver's live GPS as origin, then shows picker
  window.navigateTo=function(toCoords, toAddress){
    function _buildAndShow(fromLng, fromLat){
      var orig=fromLat+','+fromLng;
      var gUrl='',wUrl='';
      if(toCoords&&toCoords[0]&&toCoords[1]){
        var dest=toCoords[1]+','+toCoords[0];
        gUrl='https://www.google.com/maps/dir/?api=1&origin='+encodeURIComponent(orig)+
          '&destination='+encodeURIComponent(dest)+'&travelmode=driving';
        wUrl='https://waze.com/ul?ll='+dest+'&navigate=yes&from='+encodeURIComponent(orig);
      } else if(toAddress){
        gUrl='https://www.google.com/maps/dir/?api=1&origin='+encodeURIComponent(orig)+
          '&destination='+encodeURIComponent(toAddress+', South Africa')+'&travelmode=driving';
        wUrl='https://waze.com/ul?q='+encodeURIComponent(toAddress+', South Africa')+'&navigate=yes&from='+encodeURIComponent(orig);
      } else {
        toast('No destination available');return;
      }
      _showNavPicker(gUrl,wUrl);
    }
    if(navigator.geolocation){
      navigator.geolocation.getCurrentPosition(
        function(pos){driverLat=pos.coords.latitude;driverLng=pos.coords.longitude;_buildAndShow(driverLng,driverLat);},
        function(){
          if(driverLat&&driverLng){toast('⚠️ Using last known location');_buildAndShow(driverLng,driverLat);}
          else toast('❌ Could not get your location. Enable GPS and try again.');
        },
        {enableHighAccuracy:true,timeout:8000,maximumAge:5000}
      );
    } else if(driverLat&&driverLng){
      _buildAndShow(driverLng,driverLat);
    } else {
      toast('❌ GPS not available on this device');
    }
  };

  document.addEventListener('click',function(e){
    if(e.target.classList.contains('nav-start-btn')){
      var from=JSON.parse(decodeURIComponent(e.target.dataset.from||'null'));
      var to=JSON.parse(decodeURIComponent(e.target.dataset.to||'null'));
      if(from&&to)startNav(from,to);
    }
  });
})();
